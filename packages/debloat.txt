# Recommended set: safe to disable on a Mi Box 4 (Android TV 9). Each was disabled one at a time
# with health checks and survived reboots. Format: package  # reason
# Used by: scripts/04-debloat.sh     Undone by: scripts/restore.sh

# --- Ads, tracking and partner pushes ---
tv.alphonso.alphonso_eula                    # Alphonso automatic content recognition: identifies what's on screen for ad targeting
com.xm.webcontent                            # Xiaomi web content / promo pages
com.mitv.milinkservice                       # Xiaomi MiLink casting service (runs with system privileges; Chromecast does this instead)
android.autoinstalls.config.xiaomi.mibox3    # Play Auto Install config: silently installs partner apps
android.autoinstalls.config.xioami.mibox3    # same, Xiaomi's misspelt duplicate

# --- Google extras not needed on a TV ---
com.google.android.tvrecommendations         # sponsored rows for the stock home screen
com.google.android.backdrop                  # Ambient screensaver; fetches photos. scripts/05-tweaks.sh switches to the basic one
com.google.android.tv.bugreportsender        # sends bug reports to Google
com.google.android.partnersetup              # partner customisation at setup
com.google.android.marvin.talkback           # screen reader (re-enable if you need accessibility)
com.google.android.syncadapters.calendar     # calendar sync; no calendar on a TV
com.google.android.syncadapters.contacts     # contacts sync; no contacts on a TV
